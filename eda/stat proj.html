library(tidyverse)
library(lubridate)

p1_data <- atp_tennis %>%
  select(Date, Player = Player_1, Winner) %>%
  mutate(Win = ifelse(Player == Winner, 1, 0))

p2_data <- atp_tennis %>%
  select(Date, Player = Player_2, Winner) %>%
  mutate(Win = ifelse(Player == Winner, 1, 0))

all_matches <- bind_rows(p1_data, p2_data)

yearly_stats <- all_matches %>%
  mutate(Year = year(Date)) %>%  
  group_by(Year, Player) %>%
  summarise(
    Total_Matches = n(),
    Wins = sum(Win),
    Win_Percentage = Wins / Total_Matches,
    .groups = "drop"
  )

best_players <- yearly_stats %>%
  filter(Total_Matches >= 20) %>% 
  group_by(Year) %>%
  slice_max(order_by = Win_Percentage, n = 1) %>% 
  slice_max(order_by = Total_Matches, n = 1) %>%
  ungroup()

ggplot(best_players, aes(x = Year, y = Win_Percentage)) +
  geom_line(color = "gray80") + 
  geom_point(color = "steelblue", size = 3) +
  geom_text(aes(label = paste(Player, "\n", round(Win_Percentage*100, 1), "%")), 
            vjust = -0.5, size = 3, check_overlap = TRUE) +
  scale_y_continuous(labels = scales::percent_format(accuracy = 1)) +
  scale_x_continuous(breaks = unique(best_players$Year)) + 
  labs(
    title = "Best Tennis Player by Win Percentage Each Year",
    subtitle = "Minimum 20 matches played per year",
    x = "Year",
    y = "Win Percentage"
  ) +
  theme_minimal() +
  theme(axis.text.x = element_text(angle = 45, hjust = 1))

# Make Predictions on Test Data
test_data$Predicted_Prob_P1 <- predict(model, newdata = test_data, type = "response")

# Betting Strategy Simulation
betting_simulation <- test_data %>%
  mutate(
    Prediction_Correct = ifelse(
      (Predicted_Prob_P1 > 0.5 & P1_Won == 1) | (Predicted_Prob_P1 < 0.5 & P1_Won == 0), 
      1, 0
    ),
    Value_Bet_P1 = ifelse(Predicted_Prob_P1 > Implied_Prob_P1 + 0.05, 1, 0),
    Profit = case_when(
      Value_Bet_P1 == 1 & P1_Won == 1 ~ Odd_1 - 1, 
      Value_Bet_P1 == 1 & P1_Won == 0 ~ -1,        
      TRUE ~ 0                                     
    )
  )

# Analyze Results
accuracy <- mean(betting_simulation$Prediction_Correct)
total_profit <- sum(betting_simulation$Profit)
roi <- total_profit / sum(betting_simulation$Value_Bet_P1)

print(paste("Model Accuracy (Win/Loss):", round(accuracy * 100, 2), "%"))

# Data Cleaning & Feature Engineering
model_data <- atp_tennis %>%
  filter(
    !is.na(Rank_1), !is.na(Rank_2),
    !is.na(Odd_1), !is.na(Odd_2)
  ) %>%
  mutate(
    P1_Won = ifelse(Winner == Player_1, 1, 0),
    Rank_Diff = Rank_2 - Rank_1,
    Implied_Prob_P1 = 1 / Odd_1,
    Implied_Prob_P2 = 1 / Odd_2
  )

# Split Data into Train (Older) and Test 
cutoff_date <- as.Date("2023-01-01") 

train_data <- model_data %>% filter(Date < cutoff_date)
test_data  <- model_data %>% filter(Date >= cutoff_date)

print(paste("Training on", nrow(train_data), "matches."))
print(paste("Testing on", nrow(test_data), "matches."))

# Train the Model (Logistic Regression)
model <- glm(P1_Won ~ Rank_Diff, data = train_data, family = binomial)

summary(model)

# Make Predictions on Test Data
test_data$Predicted_Prob_P1 <- predict(model, newdata = test_data, type = "response")

# Betting Strategy Simulation
betting_simulation <- test_data %>%
  mutate(
    Prediction_Correct = ifelse(
      (Predicted_Prob_P1 > 0.5 & P1_Won == 1) | (Predicted_Prob_P1 < 0.5 & P1_Won == 0), 
      1, 0
    ),
    Value_Bet_P1 = ifelse(Predicted_Prob_P1 > Implied_Prob_P1 + 0.05, 1, 0),
    Profit = case_when(
      Value_Bet_P1 == 1 & P1_Won == 1 ~ Odd_1 - 1, 
      Value_Bet_P1 == 1 & P1_Won == 0 ~ -1,        
      TRUE ~ 0                                     
    )
  )

# Analyze Results
accuracy <- mean(betting_simulation$Prediction_Correct)
total_profit <- sum(betting_simulation$Profit)
roi <- total_profit / sum(betting_simulation$Value_Bet_P1)

print(paste("Model Accuracy (Win/Loss):", round(accuracy * 100, 2), "%"))
print(paste("Total Profit from Value Bets ($1/bet): $", round(total_profit, 2)))
print(paste("Return on Investment (ROI):", round(roi * 100, 2), "%"))

# Visualization: Value Bets Performance
betting_simulation %>%
  filter(Value_Bet_P1 == 1) %>%
  mutate(Cumulative_Profit = cumsum(Profit)) %>%
  ggplot(aes(x = Date, y = Cumulative_Profit)) +
  geom_line(color = "darkgreen") +
  labs(
    title = "Cumulative Profit from Betting on 'Value' (Player 1 only)",
    subtitle = "Betting $1 whenever Model Probability > Implied Odds + 5%",
    y = "Total Profit ($)",

  ) +
